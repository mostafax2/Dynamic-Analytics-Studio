php artisan analytics-suite:detect-models

| Class | Name | Table | Module | Soft Deletes | Relations |
+--------------------+-----------+----------+--------+--------------+-----------+
| App\Models\User | User | users | App | No | 3 |
| App\Models\Order | Order | orders | App | Yes | 2 |
| Modules\HR\Employee| Employee | employees| HR | Yes | 5 |

اكتمل الاكتشاف. تم العثور على 3 نموذج.
use Mostafax\AnalyticsSuite\Support\Facades\AnalyticsSuite; // اكتشاف جميع النماذج $models = AnalyticsSuite::detectModels(); foreach ($models as $model) { echo $model->name; // "User" echo $model->table; // "users" echo $model->module; // "App" أو "HR" print_r($model->relationships); // العلاقات print_r($model->columns); // الأعمدة } // توليد وِدجات تلقائية $widgets = AnalyticsSuite::detector()->generateWidgets(\App\Models\User::class); // → [KPI Card (العدد), Line Chart (النمو), Data Table]
// config/analytics-suite.php 'detection' => [ // مسارات البحث عن النماذج 'scan_paths' => [app_path('Models')], // مسار مجلد الـ Modules (nwidart/laravel-modules) 'module_paths' => [base_path('Modules')], // استثناء نماذج معينة 'excluded_models' => ['PersonalAccessToken'], // توليد وِدجات تلقائياً عند الاكتشاف 'auto_generate_widgets' => true, 'auto_generate_dashboards' => true, ],

analytics-suite:install إعداد الحزمة — يُشغَّل مرة واحدة

ينشر الإعداد والـ migrations والـ assets. يشغّل الـ migrations، يُنشئ 14 صلاحية، يكتشف النماذج، ويولّد اللوحات والوِدجات الافتراضية.

--forceاستبدال الملفات الموجودة
--skip-migrationsتخطي الـ migrations
--skip-detectionتخطي اكتشاف النماذج
--skip-defaultsتخطي توليد اللوحات الافتراضية
$ php artisan analytics-suite:install $ php artisan analytics-suite:install --force --skip-defaults
analytics-suite:detect-models اكتشاف نماذج Eloquent

يكتشف جميع نماذج Eloquent ويعرضها في جدول مع تفاصيل كاملة. يمكنه توليد وِدجات ولوحات تلقائياً.

--generate-widgetsتوليد تعريفات وِدجات لكل نموذج
--generate-dashboardsإنشاء لوحة افتراضية لكل نموذج
$ php artisan analytics-suite:detect-models $ php artisan analytics-suite:detect-models --generate-widgets --generate-dashboards

ينشئ / يحدّث الصلاحيات الـ 14 في جدول as_permissions ويسندها لدور Spatie.

FAQ

كيف أغير prefix المسارات؟

'routes' => ['prefix' => 'api/v1/analytics']

كيف أعطّل الصلاحيات في التطوير؟

ANALYTICS_ENFORCE_PERMISSIONS=false

هل يدعم nwidart/laravel-modules؟

نعم. أضف مسار الـ Modules في الإعداد:

'detection' => [ 'module_paths' => [base_path('Modules')], ]

هل يعمل بدون MongoDB؟

نعم. MongoDB اختياري. الحزمة تعمل بالكامل مع MySQL فقط. MongoDB يُفعَّل كمصدر بيانات إضافي عند الحاجة.
